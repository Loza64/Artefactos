#include <Servo.h>

// Pines del sensor ultrasónico
const int trigPin = D2;
const int echoPin = D3;
const int infrarrojo = D1;

// Pines de los LEDs
const int ledRedUltrasonico = D4;
const int ledGreenUltrasonico = D0;
const int ledRedInfrarrojo = D6;
const int ledGreenInfrarrojo = D5;

// Pines de los servomotores
const int servoUltrasonicoPin = D8;
const int servoInfrarrojoPin = D7;

Servo servoUltrasonico;
Servo servoInfrarrojo;

// Variables para los valores de distancia y luz
long duration;
int distance = 100;
int lightLevel = 1; // 1 nada, 0 carro

// Tiempos para control no bloqueante
unsigned long lastServoTimeUltrasonico = 0;
unsigned long lastServoTimeInfrarrojo = 0;
const unsigned long servoDelayUltrasonico = 6000; // 6 segundos para el ultrasonico
const unsigned long servoDelayInfrarrojo = 5000; // 5 segundos para el infrarrojo

bool ultrasonicoActive = false;
bool infrarrojoActive = false;

void setup() {
  // Configuración de pines
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(infrarrojo, INPUT);

  pinMode(ledRedUltrasonico, OUTPUT);
  pinMode(ledGreenUltrasonico, OUTPUT);
  pinMode(ledRedInfrarrojo, OUTPUT);
  pinMode(ledGreenInfrarrojo, OUTPUT);

  // LEDs por defecto
  digitalWrite(ledRedUltrasonico, HIGH);
  digitalWrite(ledGreenUltrasonico, LOW);
  digitalWrite(ledRedInfrarrojo, HIGH);
  digitalWrite(ledGreenInfrarrojo, LOW);

  // Inicializar los servomotores
  servoUltrasonico.attach(servoUltrasonicoPin);
  servoUltrasonico.write(0); // Posición inicial cerrada (0 grados)

  servoInfrarrojo.attach(servoInfrarrojoPin);
  servoInfrarrojo.write(0); // Posición inicial cerrada (0 grados)

  // Configuración del monitor serie
  Serial.begin(115200);
}

void loop() {
  // Leer la distancia del sensor ultrasónico
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  duration = pulseIn(echoPin, HIGH);
  int distance_temp = duration * 0.034 / 2; // Convertir tiempo en distancia (cm)

  // Leer el nivel del sensor infrarrojo
  int lightLevel_temp = digitalRead(infrarrojo);

  // Mostrar datos en el monitor serie
  Serial.print("Distancia: ");
  Serial.print(distance_temp);
  Serial.print(" cm | Infrarrojo: ");
  Serial.println(lightLevel_temp);

  // Control del servomotor por ultrasónico
  if (distance_temp <= 10) { // Si hay un objeto cerca
    if (!ultrasonicoActive) {
      //AQUI TE PUBLICARIA QUE HAY UN CARRO A LA PAGINA WEB ENVIANDO "1"
      //AQUI VA EL VALOR BOOLEANO QUE ME ENVIARIAS DESDE LA PAGINA WEB
      //IF(VALOR BOLEANO){codigo de abajo}
      servoUltrasonico.write(90); // Subir la barrera
      digitalWrite(ledRedUltrasonico, LOW);
      digitalWrite(ledGreenUltrasonico, HIGH);
      lastServoTimeUltrasonico = millis();
      ultrasonicoActive = true;
    }
  } else if (ultrasonicoActive && millis() - lastServoTimeUltrasonico >= servoDelayUltrasonico) {
    servoUltrasonico.write(0); // Bajar la barrera
    digitalWrite(ledRedUltrasonico, HIGH);
    digitalWrite(ledGreenUltrasonico, LOW);
    //AQUI ENVIARIA "0" A LA PAGINA WEB PARA CONFIRMARTE DE APAGAR EL BOTON DEL FORM
    ultrasonicoActive = false;
  }

  // Control del servomotor por infrarrojo
  if (lightLevel_temp == 0) { // Si se detecta un carro
    if (!infrarrojoActive) {
      servoInfrarrojo.write(90); // Subir la barrera
      digitalWrite(ledRedInfrarrojo, LOW);
      digitalWrite(ledGreenInfrarrojo, HIGH);
      lastServoTimeInfrarrojo = millis();
      infrarrojoActive = true;
    }
  } else if (infrarrojoActive && millis() - lastServoTimeInfrarrojo >= servoDelayInfrarrojo) {
    servoInfrarrojo.write(0); // Bajar la barrera
    digitalWrite(ledRedInfrarrojo, HIGH);
    digitalWrite(ledGreenInfrarrojo, LOW);
    infrarrojoActive = false;
  }
}
